Konfigurasi Database Supabase

Di Vercel, konfigurasi database akan dibaca otomatis secara aman dari **Environment Variables** (tidak perlu diisi di sini). Form di bawah ini hanya digunakan untuk simulasi pengembangan lokal (Offline Dev).

Script Inisialisasi Database (SQL DDL)

Salin script berikut dan jalankan di SQL Editor Supabase Anda untuk membuat tabel-tabel yang diperlukan: 

CREATE TABLE pembelian (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    tanggal DATE NOT NULL,
    no_seri TEXT NOT NULL UNIQUE,
    tahun INTEGER NOT NULL DEFAULT 2024,
    gramasi NUMERIC NOT NULL DEFAULT 1,
    harga_beli NUMERIC NOT NULL,
    nama_penjual TEXT NOT NULL,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now())
);

CREATE TABLE stok_manual (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    tanggal_input DATE NOT NULL,
    no_seri TEXT NOT NULL UNIQUE,
    gramasi NUMERIC NOT NULL,
    tahun INTEGER NOT NULL,
    harga_modal NUMERIC DEFAULT 0,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now())
);

CREATE TABLE penjualan (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    tanggal DATE NOT NULL,
    gramasi NUMERIC NOT NULL,
    no_seri TEXT NOT NULL,
    nama_pembeli TEXT NOT NULL,
    harga_jual NUMERIC NOT NULL,
    harga_restok NUMERIC NOT NULL,
    keuntungan NUMERIC NOT NULL,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now())
);

CREATE TABLE harga_emas (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    tanggal DATE NOT NULL,
    gram NUMERIC NOT NULL,
    harga NUMERIC NOT NULL,
    jenis_logam TEXT NOT NULL,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now())
);

CREATE TABLE settings (
    key TEXT PRIMARY KEY,
    value TEXT NOT NULL
);

INSERT INTO settings (key, value) VALUES ('pin', '1234') ON CONFLICT (key) DO NOTHING;
